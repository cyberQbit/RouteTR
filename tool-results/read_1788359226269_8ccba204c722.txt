   147→   147→  function getSmartSuggestions(state) {
   148→   148→    if (typeof PROVINCES_DATA === 'undefined') return [];
   149→   149→    
   150→   150→    const visited = new Set();

   158→   158→      }

   161→   161→    // Priority: unvisited provinces adjacent to or in same region as visited ones
   162→   162→    const suggestions = [];
   163→   163→    const suggestedPlates = new Set();
   164→   164→
   165→   165→    PROVINCES_DATA.filter(p => visited.has(p.plate)).forEach(visitedProvince => {
   166→   166→      unvisited.forEach(unvisitedProvince => {
   167→   167→        if (!suggestedPlates.has(unvisitedProvince.plate) && 
   168→   168→            (unvisitedProvince.region === visitedProvince.region || 
   169→   169→             isAdjacentRegion(visitedProvince.region, unvisitedProvince.region))) {
   170→   170→          suggestions.push({
   171→   171→            province: unvisitedProvince,
   172→   172→            reason: `${visitedProvince.name}'nin yanında`,
   173→   173→          });
   174→   174→          suggestedPlates.add(unvisitedProvince.plate);
   175→   175→        }
   176→   176→      });
   177→   177→    });
   178→   178→
   179→   179→    return suggestions.slice(0, 3);
   180→   180→  }

   299→   299→  // Score Calculation
   300→   300→  function calculateProvinceScore(plate) {
   301→   301→    const data = travelState[plate];
   302→   302→    const pMeta = PROVINCES_DATA.find(p => p.plate === plate);
   303→   303→    if (!data || !pMeta || data.status === 'unvisited') return 0;
   304→   304→
   305→   305→    const totalDistricts = pMeta.districts.length || 1;
   306→   306→    const visitedDistrictsCount = (data.visitedDistricts || []).length;
   307→   307→    const districtRatio = Math.min(1, visitedDistrictsCount / totalDistricts);
   308→   308→
   309→   309→    const allPois = [...pMeta.pois, ...(data.customPois || [])];
   310→   310→    const totalPois = allPois.length || 1;
   311→   311→    const visitedPoisCount = (data.visitedPois || []).length;
   312→   312→    const poiRatio = Math.min(1, visitedPoisCount / totalPois);
   313→   313→
   314→   314→    if (data.status === 'transit') {
   315→   315→      return Math.min(30, Math.round(15 + districtRatio * 10 + poiRatio * 5));
   316→   316→    }
   317→   317→
   318→   318→    if (data.status === 'lived') {
   319→   319→      return Math.min(100, Math.round(70 + districtRatio * 15 + poiRatio * 15));
   320→   320→    }
   321→   321→
   322→   322→    return Math.min(100, Math.round(30 + districtRatio * 40 + poiRatio * 30));
   323→   323→  }

   346→   346→  function checkProvinceMatch(p) {
   347→   347→    const pData = travelState[p.plate];
   348→   348→    const normQuery = normalizeText(searchQuery);
   349→   349→
   350→   350→    let matchesSearch = true;
   351→   351→    if (normQuery) {
   352→   352→      const matchName = normalizeText(p.name).includes(normQuery);
   353→   353→      const matchPlate = p.plate.includes(normQuery);
   354→   354→      const matchDistrict = (p.districts || []).some(d => normalizeText(d).includes(normQuery));
   355→   355→      const matchPoi = (p.pois || []).some(poi => normalizeText(poi).includes(normQuery));
   356→   356→      const matchCustom = (pData.customPois || []).some(c => normalizeText(c).includes(normQuery));
   357→   357→      matchesSearch = matchName || matchPlate || matchDistrict || matchPoi || matchCustom;
   358→   358→    }
   359→   359→
   360→   360→    const matchesRegion = currentFilterRegion === 'all' || p.region === currentFilterRegion;
   361→   361→    
   362→   362→    let matchesStatus = true;
   363→   363→    if (currentFilterStatus === 'visited') matchesStatus = pData.status === 'visited';
   364→   364→    else if (currentFilterStatus === 'transit') matchesStatus = pData.status === 'transit';
   365→   365→    else if (currentFilterStatus === 'lived') matchesStatus = pData.status === 'lived';
   366→   366→    else if (currentFilterStatus === 'unvisited') matchesStatus = pData.status === 'unvisited';
   367→   367→
   368→   368→    return matchesSearch && matchesRegion && matchesStatus;
