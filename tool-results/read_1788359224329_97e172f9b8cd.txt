   164→
   165→    PROVINCES_DATA.filter(p => visited.has(p.plate)).forEach(visitedProvince => {
   166→      unvisited.forEach(unvisitedProvince => {
   167→        if (!suggestedPlates.has(unvisitedProvince.plate) && 
   168→            (unvisitedProvince.region === visitedProvince.region || 
   169→             isAdjacentRegion(visitedProvince.region, unvisitedProvince.region))) {
   170→          suggestions.push({
   171→            province: unvisitedProvince,
   172→            reason: `${visitedProvince.name}'nin yanında`,
   173→          });
   174→          suggestedPlates.add(unvisitedProvince.plate);
   175→        }
   176→      });
   177→    });

   299→  // Score Calculation
   300→  function calculateProvinceScore(plate) {
   301→    const data = travelState[plate];
   302→    const pMeta = PROVINCES_DATA.find(p => p.plate === plate);
   303→    if (!data || !pMeta || data.status === 'unvisited') return 0;
   304→
   305→    const totalDistricts = pMeta.districts.length || 1;
   306→    const visitedDistrictsCount = (data.visitedDistricts || []).length;
   307→    const districtRatio = Math.min(1, visitedDistrictsCount / totalDistricts);
   308→
   309→    const allPois = [...pMeta.pois, ...(data.customPois || [])];
   310→    const totalPois = allPois.length || 1;
   311→    const visitedPoisCount = (data.visitedPois || []).length;
   312→    const poiRatio = Math.min(1, visitedPoisCount / totalPois);
   313→
   314→    if (data.status === 'transit') {
   315→      return Math.min(30, Math.round(15 + districtRatio * 10 + poiRatio * 5));
   316→    }
   317→
   318→    if (data.status === 'lived') {
   319→      return Math.min(100, Math.round(70 + districtRatio * 15 + poiRatio * 15));
   320→    }
   321→
   322→    return Math.min(100, Math.round(30 + districtRatio * 40 + poiRatio * 30));
   323→  }

   346→  function checkProvinceMatch(p) {
   347→    const pData = travelState[p.plate];
   348→    const normQuery = normalizeText(searchQuery);
   349→
   350→    let matchesSearch = true;
   351→    if (normQuery) {
   352→      const matchName = normalizeText(p.name).includes(normQuery);
   353→      const matchPlate = p.plate.includes(normQuery);
   354→      const matchDistrict = (p.districts || []).some(d => normalizeText(d).includes(normQuery));
   355→      const matchPoi = (p.pois || []).some(poi => normalizeText(poi).includes(normQuery));
   356→      const matchCustom = (pData.customPois || []).some(c => normalizeText(c).includes(normQuery));
   357→      matchesSearch = matchName || matchPlate || matchDistrict || matchPoi || matchCustom;
   358→    }
   359→
   360→    const matchesRegion = currentFilterRegion === 'all' || p.region === currentFilterRegion;
   361→    
   362→    let matchesStatus = true;
   363→    if (currentFilterStatus === 'visited') matchesStatus = pData.status === 'visited';
   364→    else if (currentFilterStatus === 'transit') matchesStatus = pData.status === 'transit';
   365→    else if (currentFilterStatus === 'lived') matchesStatus = pData.status === 'lived';
   366→    else if (currentFilterStatus === 'unvisited') matchesStatus = pData.status === 'unvisited';
   367→
   368→    return matchesSearch && matchesRegion && matchesStatus;
   369→  }

   453→  // 2. Try to Load External SVG file if available (e.g. TR-adm2-with-city-borders.svg)
   454→  async function tryLoadExternalSvgMap() {
   455→    const candidateFiles = [
   456→      'TR-adm2-with-city-borders.svg',
   457→      'TR-adm1.svg',
   458→      'turkey.svg'
   459→    ];
   460→
   461→    let loadedSvgText = null;
   462→    for (const filename of candidateFiles) {

   464→        const resp = await fetch(filename);
   465→        if (resp.ok) {
   466→          loadedSvgText = await resp.text();
   467→          break;
   468→        }
   469→      } catch (e) {}
   470→    }
   471→
   472→    if (loadedSvgText) {
   473→      const parser = new DOMParser();
   474→      const doc = parser.parseFromString(loadedSvgText, 'image/svg+xml');
   475→      const svgRoot = doc.querySelector('svg');
   476→      if (svgRoot) {
   477→        svgMapContainer.setAttribute('viewBox', svgRoot.getAttribute('viewBox') || '0 0 1007 437');
   478→        svgMapContainer.innerHTML = svgRoot.innerHTML;
   479→        setupSvgMapEvents();
   480→        updateMapFills();

   482→    }
